{Ejercicio 2: Realice un programa en el que 3 robots realizan una escalera de 4 escalones cada uno. Todos los escalones tienen un ancho fijo de 1, y un alto aleatorio entre 1 y 5. Al finalizar el recorrido, cada robot deberá enviar al robot jefe la cantidad de escalones que tenían más flores que papeles. Una vez que los tres robots finalizaron, el robot jefe deberá informar la suma de las cantidades enviadas por los 3 robots.
- El robot jefe inicia en la esquina (1,1)
- El robot 1 inicia en la esquina (2,1)
- El robot 2 inicia en la esquina (20,1)
- El robot 3 inicia en la esquina (40,1)
}

programa ejemplo
procesos
  proceso hacerEscalon(ES f: numero; ES p: numero)
  variables
    alto: numero
  comenzar
    Random(alto, 1,5)
    repetir alto
      mientras HayPapelEnLaEsquina
        tomarPapel
        p:= p + 1
      mientras HayFlorEnLaEsquina
        tomarFlor
        f:= f + 1
      mover
    derecha
    mientras HayPapelEnLaEsquina
      tomarPapel
      p:= p + 1
    mientras HayFlorEnLaEsquina
      tomarFlor
      f:= f + 1
    mover
    repetir 3
      derecha
  fin
  
  proceso hacerEscalones
  variables
    f, p: numero
    i: numero
  comenzar
    i:= 0
    repetir 4
      f:= 0
      p:= 0
      hacerEscalon(f, p)
      si (f > p)
        i:= i + 1
    EnviarMensaje(i, jefe)
  fin
areas
  areaJefe: AreaP (1,1,1,1)
  area1: AreaP (2,1,6,21)
  area2: AreaP (20,1,24,21)
  area3: AreaP (40,1,44,21)
robots
  robot robot1
  comenzar
    hacerEscalones
  fin
  
  robot robot2
  variables
    aux: numero
    total: numero
  comenzar  
    total:= 0
    repetir 3
      RecibirMensaje(aux, *)
      total:= total + aux
    Informar(total)
    
  fin
  
variables
  r1: robot1
  r2: robot1
  r3: robot1
  jefe: robot2
comenzar
  AsignarArea(jefe, areaJefe)
  AsignarArea(r1, area1)
  AsignarArea(r2, area2)
  AsignarArea(r3, area3)
  Iniciar(jefe, 1,1)
  Iniciar(r1, 2,1)
  Iniciar(r2, 20,1)
  Iniciar(r3, 40,1)
fin
